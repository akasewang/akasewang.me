When pages come from MDX, maintaining a second list for `sitemap.xml` or `feed.xml` creates metadata that can drift out of date. The safer design is to let the pages, sitemap and feed read the same content index.

The standalone example below uses two small packages:

```bash
npm install gray-matter rss server-only
npm install --save-dev @types/rss
```

## Build one content adapter

Start with a server-only function that reads published MDX frontmatter. This is the only part that should change if the content later moves to a CMS or database.

```ts title="lib/posts.ts"
import 'server-only'

import { readdir, readFile } from 'node:fs/promises'
import path from 'node:path'
import matter from 'gray-matter'

const postsDirectory = path.join(process.cwd(), 'content', 'posts')

export type PostSummary = {
  slug: string
  title: string
  excerpt: string
  publishedAt: string
}

function asIsoDate(value: unknown): string | null {
  const date = value instanceof Date ? value : new Date(String(value))
  return Number.isNaN(date.getTime()) ? null : date.toISOString()
}

export async function getPublishedPosts(): Promise<PostSummary[]> {
  // My site uses an MDX content manager; this standalone version reads files directly.
  const filenames = (await readdir(postsDirectory)).filter((name) => name.endsWith('.mdx'))

  const posts = await Promise.all(
    filenames.map(async (filename) => {
      const source = await readFile(path.join(postsDirectory, filename), 'utf8')
      const { data } = matter(source)
      const publishedAt = asIsoDate(data.date)

      if (data.draft || !publishedAt || typeof data.title !== 'string') return null

      return {
        slug: path.basename(filename, '.mdx'),
        title: data.title,
        excerpt: typeof data.excerpt === 'string' ? data.excerpt : '',
        publishedAt,
      }
    }),
  )

  return posts
    .filter((post): post is PostSummary => post !== null)
    .toSorted((a, b) => Date.parse(b.publishedAt) - Date.parse(a.publishedAt))
}
```

The adapter validates the values consumed by public metadata instead of assuming every frontmatter field is usable. It also keeps filesystem access out of client bundles with `server-only`.

## Use Next.js sitemap metadata

Next.js can serialize a typed sitemap without hand-building XML. Add `SITE_URL=https://example.com` to the environment, then return static routes and content routes from `app/sitemap.ts`:

```ts title="app/sitemap.ts"
import type { MetadataRoute } from 'next'
import { getPublishedPosts } from '../lib/posts'

function getSiteUrl(): string {
  const value = process.env.SITE_URL
  if (!value) throw new Error('SITE_URL is required')
  return value.replace(/\/$/, '')
}

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const siteUrl = getSiteUrl()
  const posts = await getPublishedPosts()

  const staticPages: MetadataRoute.Sitemap = [
    { url: siteUrl, changeFrequency: 'weekly', priority: 1 },
    { url: `${siteUrl}/posts`, changeFrequency: 'daily', priority: 0.9 },
    { url: `${siteUrl}/about`, changeFrequency: 'monthly', priority: 0.6 },
  ]

  const postPages: MetadataRoute.Sitemap = posts.map((post) => ({
    url: `${siteUrl}/posts/${encodeURIComponent(post.slug)}`,
    lastModified: post.publishedAt,
    changeFrequency: 'weekly',
    priority: 0.8,
  }))

  return [...staticPages, ...postPages]
}
```

The metadata file is the standard choice when a sitemap only needs the fields Next.js already supports. It also avoids manual XML escaping and automatically serves the result from `/sitemap.xml`.

Only include URLs that actually resolve locally. If some content entries use an external canonical URL, filter them out before creating `postPages`.

## Build RSS with a serializer

RSS contains user-visible text and more envelope metadata, so a serializer is preferable to string interpolation. A Route Handler can reuse the same adapter:

```ts title="app/feed.xml/route.ts"
import RSS from 'rss'
import { getPublishedPosts } from '../../lib/posts'

function getSiteUrl(): string {
  const value = process.env.SITE_URL
  if (!value) throw new Error('SITE_URL is required')
  return value.replace(/\/$/, '')
}

export async function GET(): Promise<Response> {
  try {
    const siteUrl = getSiteUrl()
    const posts = await getPublishedPosts()
    const feed = new RSS({
      title: 'Example Blog',
      description: 'Articles about design and development',
      site_url: siteUrl,
      feed_url: `${siteUrl}/feed.xml`,
      language: 'en',
    })

    for (const post of posts) {
      feed.item({
        title: post.title,
        description: post.excerpt,
        url: `${siteUrl}/posts/${encodeURIComponent(post.slug)}`,
        guid: `${siteUrl}/posts/${encodeURIComponent(post.slug)}`,
        date: new Date(post.publishedAt),
      })
    }

    return new Response(feed.xml({ indent: true }), {
      headers: {
        'Content-Type': 'application/rss+xml; charset=utf-8',
        'Cache-Control': 'public, s-maxage=3600, stale-while-revalidate=86400',
      },
    })
  } catch (error) {
    console.error('Unable to generate RSS feed', error)
    return new Response('Unable to generate feed', { status: 500 })
  }
}
```

Returning a clear `500` is preferable to serving partial XML. Feed readers can retry a failed response, while malformed XML can be cached or interpreted inconsistently.

`s-maxage` controls shared caches rather than the reader's browser cache. The explicit `stale-while-revalidate` window lets a CDN serve a known feed while it refreshes the document, subject to the hosting platform's support for that directive.

The important boundary is not MDX specifically. One validated content adapter owns publication metadata; the page, sitemap and feed each transform that same data for their own protocol.
